        // --- VARIABLEN ---
        let data = { groups: [] };
        let nextGroupId = 1;
        let nextPlayerId = 1;
        let saveTimer = null;
        let parsedImport = []; 

        let knownStats = {}; 
        let checkingGroups = {};

        let currentExportData = {};

        loadData();

        function loadData() {
            fetch('/api/players')
                .then(r => r.json())
                .then(d => {
                    data = d || { groups: [] };
                    if (!data.groups) data.groups = [];
                    data.groups.forEach(g => {
                        if (g.id >= nextGroupId) nextGroupId = g.id + 1;
                        g.players.forEach(p => {
                            if (p.id >= nextPlayerId) nextPlayerId = p.id + 1;
                        });
                    });
                    renderGroups();
                })
                .catch(e => {
                    console.error('Load error:', e);
                    renderGroups();
                });
        }

        function hasAnyStats(g) {
            return g.players.some(p => knownStats[p.id] === true);
        }

        async function checkGroupStats(gid) {
            if (checkingGroups[gid]) return; 
            checkingGroups[gid] = true;
            
            let grp = data.groups.find(g => g.id === gid);
            if (!grp) {
                checkingGroups[gid] = false;
                return;
            }

            let pidsToCheck = grp.players.filter(p => knownStats[p.id] === undefined).map(p => p.id);

            if (pidsToCheck.length > 0) {
                try {
                    let r = await fetch('/api/stats/check?pids=' + pidsToCheck.join(','));
                    if (r.ok) {
                        let result = await r.json();
                        for (let pid in result) {
                            knownStats[pid] = result[pid];
                        }
                    }
                } catch(e) {
                    console.error("Fehler beim Abfragen der Stats:", e);
                }
            }

            let grpHasStats = false;
            for (let p of grp.players) {
                if (knownStats[p.id]) {
                    grpHasStats = true;
                    let pBtn = document.getElementById(`btn-stat-${p.id}`);
                    if (pBtn) pBtn.style.display = 'inline-block';
                }
            }
            
            if (grpHasStats) {
                let gBtn = document.getElementById(`btn-grp-stat-${gid}`);
                if (gBtn) gBtn.style.display = 'inline-block';
            }
            
            checkingGroups[gid] = false;
        }

        function sanitize(str, maxLen) {
            if (typeof str !== 'string') return '';
            return str.replace(/[{}\[\]"'\\<>&`]/g, '').replace(/\s+/g, ' ').trim().substring(0, maxLen || 20);
        }
        function sanitizeGroupName(str) { return sanitize(str, 25); }
        function sanitizePlayerName(str) { return sanitize(str, 20); }

            let container = document.getElementById('groupList');

            if (data.groups.length === 0) {
                container.innerHTML = `
                    <div class="empty-state">
                        <span class="icon">👥</span>
                        <p data-i18n="pm_empty">No groups yet. Add a group or import players!</p>
                    </div>`;
                applyTranslations();
                return;
            }

            let html = '';
            data.groups.forEach((g, gi) => {
                let isOpen = document.querySelector(`.grp-card[data-gid="${g.id}"]`)?.classList.contains('open');

        function escHTML(s) {
            return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
        }

        function toggleGroup(gid) {
            let card = document.querySelector(`.grp-card[data-gid="${gid}"]`);
            if (card) {
                card.classList.toggle('open');
                if (card.classList.contains('open')) {
                    checkGroupStats(gid);
                }
            }
        }

        function addGroup() {
            let name = prompt(getTranslation('pm_prompt_group') || 'Group name:');
            if (!name) return;
            name = sanitizeGroupName(name);
            if (!name) return;
            data.groups.push({ id: nextGroupId++, name: name, players: [] });
            renderGroups();
            scheduleSave();
            setTimeout(() => {
                let card = document.querySelector(`.grp-card[data-gid="${nextGroupId - 1}"]`);
                if (card) card.classList.add('open');
            }, 50);
        }

        function deleteGroup(gid) {
            let grp = data.groups.find(g => g.id === gid);
            if (!grp) return;
            let msg = (getTranslation('pm_confirm_del_grp') || 'Delete group "{name}" and all {n} players?').replace('{name}', grp.name).replace('{n}', grp.players.length);
            if (!confirm(msg)) return;
            
            grp.players.forEach(p => { 
                fetch('/api/stats/player_del?pid=' + p.id).catch(e=>{}); 
                delete knownStats[p.id];
            });

            data.groups = data.groups.filter(g => g.id !== gid);
            renderGroups();
            scheduleSave();
        }

        function renameGroup(gid, newName) {
            newName = sanitizeGroupName(newName);
            let grp = data.groups.find(g => g.id === gid);
            if (grp && newName) { grp.name = newName; scheduleSave(); }
        }

        function addPlayerFromInput(gid) {
            let inp = document.getElementById('newPlayer-' + gid);
            let name = sanitizePlayerName(inp.value);
            if (!name) return;

            let grp = data.groups.find(g => g.id === gid);
            if (!grp) return;

            if (grp.players.some(p => p.name.toLowerCase() === name.toLowerCase())) {
                showToast('⚠️ ' + name + ' exists!', true);
                return;
            }

            let newId = nextPlayerId++;
            grp.players.push({ id: newId, name: name });
            knownStats[newId] = false; 

            inp.value = '';
            renderGroups();
            scheduleSave();

            setTimeout(() => {
                let card = document.querySelector(`.grp-card[data-gid="${gid}"]`);
                if (card) card.classList.add('open');
                let newInp = document.getElementById('newPlayer-' + gid);
                if (newInp) newInp.focus();
            }, 50);
        }

        function deletePlayer(gid, pid) {
            let grp = data.groups.find(g => g.id === gid);
            if (!grp) return;
            grp.players = grp.players.filter(p => p.id !== pid);
            fetch('/api/stats/player_del?pid=' + pid).catch(e=>{});
            delete knownStats[pid];
            
            renderGroups();
            setTimeout(() => {
                let card = document.querySelector(`.grp-card[data-gid="${gid}"]`);
                if (card) card.classList.add('open');
            }, 50);
            scheduleSave();
        }

        function renamePlayer(gid, pid, newName) {

            let grp = data.groups.find(g => g.id === gid);
            if (!grp) return;
            let player = grp.players.find(p => p.id === pid);
            if (player && newName) { player.name = newName; scheduleSave(); }

        function parsePlayerCSVForExport(playerName, text) {
            let lines = text.split(/\r?\n/).filter(l => l.trim().length > 0);
            lines.forEach(line => {
                let parts = line.split(';');
                if (parts.length >= 3) {
                    let ts = parseInt(parts[0]);
                    let gId = parseInt(parts[1]);
                    let score = parseInt(parts[2]);
                    
                    let ex1 = parts.length >= 4 ? parts[3] : "0"; 
                    let ex2 = parts.length >= 5 ? parts[4] : "0"; 
                    let ex3 = parts.length >= 6 ? parts[5] : "0"; 
                    let ex4 = parts.length >= 7 ? parts[6] : "0"; 
                    let ex5 = parts.length >= 8 ? parts[7] : "0"; 

                    if (!currentExportData[gId]) currentExportData[gId] = [];
                    currentExportData[gId].push({ name: playerName, ts, score, ex1, ex2, ex3, ex4, ex5 });
                }
            });
        }

        function formatValueForExport(val, type) {

            if (type === "ms_to_s1") return (parseInt(val) / 1000).toFixed(1).replace('.', ',');
            if (type === "bool") return parseInt(val) === 1 ? "1" : "0";
            return val;
        }

        function downloadGameCSV(gId) {
            let info = GAME_DICT[gId] || { name: "Game " + gId, lblScore: "Score", lblExtra: "Extra", fmtE: "num" };
            let entries = currentExportData[gId] || [];
            entries.sort((a, b) => b.ts - a.ts);

            let txtName = t('ps_name', 'Name');
            let txtDate = t('ps_date', 'Datum');
            let txtTime = t('ps_time_h', 'Uhrzeit');

            // Dynamischer CSV Header
            let header = `${txtName};${txtDate};${txtTime};${info.lblScore}`;
            if (info.fmtE !== "none") header += `;${info.lblExtra}`;
            if (info.lblEx2) header += `;${info.lblEx2}`;
            if (info.lblEx3) header += `;${info.lblEx3}`;
            if (info.lblEx4) header += `;${info.lblEx4}`;
            if (info.lblEx5) header += `;${info.lblEx5}`;
            let lines = [header];

            entries.forEach(e => {
                let d = new Date(e.ts);
                let dateStr = d.toLocaleDateString('de-DE');
                let timeStr = d.toLocaleTimeString('de-DE', {hour: '2-digit', minute:'2-digit', second:'2-digit'});
                
                let line = `${e.name};${dateStr};${timeStr};${formatValueForExport(e.score, info.fmtS)}`;
                
                // Dynamische CSV Spalten
                if (info.fmtE !== "none") line += `;${formatValueForExport(e.ex1, info.fmtE)}`;
                if (info.lblEx2) line += `;${formatValueForExport(e.ex2, info.fmtEx2)}`;
                if (info.lblEx3) line += `;${formatValueForExport(e.ex3, info.fmtEx3)}`;
                if (info.lblEx4) line += `;${formatValueForExport(e.ex4, info.fmtEx4)}`;
                if (info.lblEx5) line += `;${formatValueForExport(e.ex5, info.fmtEx5)}`;
                
                lines.push(line);
            });

            let csvStr = lines.join('\r\n');
            let blob = new Blob(['\uFEFF' + csvStr], { type: 'text/csv;charset=utf-8' });
            let url = URL.createObjectURL(blob);
            let safeGrpName = currentExportGroupName.replace(/[^a-zA-Z0-9]/g, '_');
            let safeGameName = info.name.replace(/[^a-zA-Z0-9]/g, '_');
            let stamp = new Date().toISOString().split('T')[0];
            
            let a = document.createElement('a');
            a.href = url;
            a.download = `Stats_${safeGrpName}_${safeGameName}_${stamp}.csv`;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
        }

            parsedImport.forEach(e => { html += `<tr><td class="col-grp">${escHTML(e.group)}</td><td class="col-name">${escHTML(e.name)}</td></tr>`; });
            html += `<tr><td colspan="2" style="color:var(--success);padding-top:8px;font-weight:bold;">✓ ${parsedImport.length} entries</td></tr>`;
            table.innerHTML = html;
            preview.style.display = 'block';
            btn.disabled = false;
        }

        function parseImportText(text) {
            if (!text || !text.trim()) return [];
            if (text.charCodeAt(0) === 0xFEFF) text = text.substring(1);
            let lines = text.split(/\r?\n/).map(l => l.trim()).filter(l => l.length > 0);
            if (lines.length === 0) return [];
            let firstLower = lines[0].toLowerCase();
            if (firstLower.includes('name') && (firstLower.includes('group') || firstLower.includes('gruppe') || firstLower.includes('class') || firstLower.includes('klasse') || firstLower.includes('team'))) { lines.shift(); }
            
            let sample = lines.slice(0, 10).join('\n');
            let delim = null;
            if (sample.includes('\t')) delim = '\t';
            else if (sample.includes(';')) delim = ';';
            else {
                let commaLines = lines.filter(l => l.includes(',')).length;
                if (commaLines > lines.length * 0.5) delim = ',';
            }

            let entries = [];
            for (let line of lines) {
                if (delim) {
                    let parts = splitRespectingQuotes(line, delim);
                    if (parts.length >= 2) {
                        let group = sanitizeGroupName(parts[0]);
                        let name = sanitizePlayerName(parts[1]);
                        if (name) entries.push({ group: group || 'Import', name });
                    } else if (parts.length === 1) {
                        let name = sanitizePlayerName(parts[0]);
                        if (name) entries.push({ group: 'Import', name });
                    }
                } else {
                    let name = sanitizePlayerName(line);
                    if (name) entries.push({ group: 'Import', name });
                }
            }
            return entries;
        }

        function splitRespectingQuotes(line, delim) {
            let parts = []; let current = ''; let inQuotes = false;
            for (let i = 0; i < line.length; i++) {
                let ch = line[i];
                if (ch === '"') { inQuotes = !inQuotes; continue; }
                if (ch === delim && !inQuotes) { parts.push(current.trim()); current = ''; continue; }
                current += ch;
            }
            parts.push(current.trim());
            return parts;
        }

        function executeImport() {
            if (parsedImport.length === 0) return;
            let added = 0;
            parsedImport.forEach(entry => {
                let grp = data.groups.find(g => g.name.toLowerCase() === entry.group.toLowerCase());
                if (!grp) { grp = { id: nextGroupId++, name: entry.group, players: [] }; data.groups.push(grp); }
                if (!grp.players.some(p => p.name.toLowerCase() === entry.name.toLowerCase())) {
                    grp.players.push({ id: nextPlayerId++, name: entry.name }); added++;
                }
            });
            closeImport();
            renderGroups();
            scheduleSave();
            showToast(`✓ ${added} players imported`);
        }

/*
 * PLAYER PICKER - Wiederverwendbare Spieler-Auswahl-Komponente
 * Version: 1.0
 * 
 * Einbindung in jede *_names.html:
 *   <script src="player_picker.js"></script>
 *   <button onclick="openPlayerPicker(numSlots, onPlayersSelected)" class="btn-primary">👥 Spieler laden</button>
 *
 * Callback: onPlayersSelected([{id: 101, name: "Lena M."}, ...])
 */

const PlayerPicker = (function() {

    let _overlay = null;
    let _data = null;       // { groups: [...] }
    let _numSlots = 0;
    let _callback = null;
    let _selected = [];     // [{id, name}, ...]
    let _currentGroupId = null;

    // ==================== SANITIZATION ====================
    function sanitize(str, maxLen) {
        if (typeof str !== 'string') return '';
        return str
            .replace(/[{}\[\]"'\\<>&`]/g, '')
            .replace(/\s+/g, ' ')
            .trim()
            .substring(0, maxLen || 20);
    }

    // ==================== UI ERSTELLEN ====================
    function createOverlay() {
        if (_overlay) _overlay.remove();

        _overlay = document.createElement('div');
        _overlay.id = 'playerPickerOverlay';
        _overlay.innerHTML = `
            <div class="pp-sheet">
                <div class="pp-header">
                    <span class="pp-title">👥 ${_tr('pp_title')}</span>
                    <button class="pp-close" onclick="PlayerPicker.close()">✕</button>
                </div>
                <div class="pp-body">
                    <div class="pp-group-bar">
                        <select id="ppGroupSelect" onchange="PlayerPicker.selectGroup(this.value)">
                            <option value="">— ${_tr('pp_select_group')} —</option>
                        </select>
                    </div>
                    <div class="pp-counter" id="ppCounter"></div>
                    <div class="pp-list" id="ppPlayerList">
                        <div class="pp-hint">${_tr('pp_hint')}</div>
                    </div>
                </div>
                <div class="pp-footer">
                    <button class="pp-btn-cancel" onclick="PlayerPicker.close()">${_tr('btn_cancel')}</button>
                    <button class="pp-btn-apply" id="ppApplyBtn" onclick="PlayerPicker.apply()" disabled>${_tr('pp_apply')}</button>
                </div>
            </div>
        `;
        document.body.appendChild(_overlay);

        // Animation: einblenden
        requestAnimationFrame(() => _overlay.classList.add('pp-visible'));
    }

    // ==================== DATEN LADEN ====================
    function loadData() {
        fetch('/api/players')
            .then(r => r.json())
            .then(data => {
                _data = data;
                populateGroups();
                // Letzte Gruppe wiederherstellen
                let lastGrp = sessionStorage.getItem('pp_lastGroup');
                if (lastGrp) {
                    let sel = document.getElementById('ppGroupSelect');
                    sel.value = lastGrp;
                    if (sel.value === lastGrp) selectGroup(lastGrp);
                }
            })
            .catch(e => {
                console.error('PlayerPicker: Load failed', e);
                document.getElementById('ppPlayerList').innerHTML = 
                    '<div class="pp-hint" style="color:var(--danger)">Fehler beim Laden!</div>';
            });
    }

    function populateGroups() {
        let sel = document.getElementById('ppGroupSelect');
        if (!_data || !_data.groups) return;
        _data.groups.forEach(g => {
            let opt = document.createElement('option');
            opt.value = g.id;
            opt.textContent = `${g.name} (${g.players.length})`;
            sel.appendChild(opt);
        });
    }

    // ==================== GRUPPE AUSWÄHLEN ====================
    function selectGroup(groupId) {
        _currentGroupId = parseInt(groupId);
        _selected = [];
        sessionStorage.setItem('pp_lastGroup', groupId);

        let group = _data.groups.find(g => g.id === _currentGroupId);
        let container = document.getElementById('ppPlayerList');

        if (!group || !group.players.length) {
            container.innerHTML = '<div class="pp-hint">' + _tr('pp_empty') + '</div>';
            updateCounter();
            return;
        }

        // Letzte Auswahl wiederherstellen
        let lastSel = JSON.parse(sessionStorage.getItem('pp_lastSelected_' + groupId) || '[]');

        let html = '';
        group.players.forEach(p => {
            let checked = lastSel.includes(p.id);
            if (checked) _selected.push({id: p.id, name: p.name});
            html += `
                <label class="pp-player ${checked ? 'pp-checked' : ''}" id="pp-p-${p.id}">
                    <input type="checkbox" ${checked ? 'checked' : ''} 
                           onchange="PlayerPicker.togglePlayer(${p.id}, '${sanitize(p.name, 20).replace(/'/g, "\\'")}', this.checked)">
                    <span class="pp-check">${checked ? '✓' : ''}</span>
                    <span class="pp-name">${sanitize(p.name, 20)}</span>
                </label>`;
        });

        // "Alle auswählen" Toggle
        html = `<label class="pp-select-all" onclick="PlayerPicker.toggleAll()">
                    <span id="ppToggleAllTxt">${_tr('pp_select_all')}</span>
                </label>` + html;

        container.innerHTML = html;
        updateCounter();
    }

    // ==================== SPIELER TOGGLEN ====================
    function togglePlayer(id, name, checked) {
        if (checked) {
            if (!_selected.find(s => s.id === id)) {
                _selected.push({id, name});
            }
        } else {
            _selected = _selected.filter(s => s.id !== id);
        }

        // Visual Feedback
        let label = document.getElementById('pp-p-' + id);
        if (label) {
            label.classList.toggle('pp-checked', checked);
            label.querySelector('.pp-check').textContent = checked ? '✓' : '';
        }

        updateCounter();
    }

    function toggleAll() {
        let group = _data.groups.find(g => g.id === _currentGroupId);
        if (!group) return;

        let allSelected = _selected.length === group.players.length;
        
        if (allSelected) {
            // Deselect all
            _selected = [];
            group.players.forEach(p => {
                let cb = document.querySelector(`#pp-p-${p.id} input`);
                if (cb) cb.checked = false;
                let label = document.getElementById('pp-p-' + p.id);
                if (label) { label.classList.remove('pp-checked'); label.querySelector('.pp-check').textContent = ''; }
            });
        } else {
            // Select all
            _selected = group.players.map(p => ({id: p.id, name: p.name}));
            group.players.forEach(p => {
                let cb = document.querySelector(`#pp-p-${p.id} input`);
                if (cb) cb.checked = true;
                let label = document.getElementById('pp-p-' + p.id);
                if (label) { label.classList.add('pp-checked'); label.querySelector('.pp-check').textContent = '✓'; }
            });
        }
        updateCounter();
    }

    // ==================== COUNTER + VALIDIERUNG ====================
    function updateCounter() {
        let el = document.getElementById('ppCounter');
        let btn = document.getElementById('ppApplyBtn');
        let n = _selected.length;

        let cls = '';
        if (n === 0) { cls = 'pp-cnt-empty'; }
        else if (n > _numSlots) { cls = 'pp-cnt-warn'; }
        else if (n === _numSlots) { cls = 'pp-cnt-ok'; }
        else { cls = 'pp-cnt-partial'; }

        let warnText = '';
        if (n > _numSlots) {
            warnText = `<div class="pp-warn">⚠️ ${_tr('pp_too_many').replace('{n}', n - _numSlots)}</div>`;
        }

        el.innerHTML = `<span class="${cls}">✓ ${n} / ${_numSlots}</span>${warnText}`;
        btn.disabled = (n === 0);
    }

    // ==================== APPLY ====================
    function apply() {
        // Auswahl in sessionStorage merken
        if (_currentGroupId) {
            sessionStorage.setItem('pp_lastSelected_' + _currentGroupId, 
                JSON.stringify(_selected.map(s => s.id)));
        }

        // Auf numSlots begrenzen
        let result = _selected.slice(0, _numSlots);

        close();
        if (_callback) _callback(result);
    }

    // ==================== CLOSE ====================
    function close() {
        if (_overlay) {
            _overlay.classList.remove('pp-visible');
            setTimeout(() => { if (_overlay) _overlay.remove(); _overlay = null; }, 300);
        }
    }

    // ==================== i18n HELPER ====================
    function _tr(key) {
        if (typeof getTranslation === 'function') return getTranslation(key);
        // Fallback Englisch
        const fb = {
            'pp_title': 'Select Players',
            'pp_select_group': 'Select Group',
            'pp_hint': 'Select a group to see players',
            'pp_empty': 'No players in this group',
            'pp_select_all': '☐ Select All / Deselect All',
            'pp_apply': '✓ APPLY',
            'pp_too_many': '{n} too many — extras will be skipped',
            'btn_cancel': 'Cancel'
        };
        return fb[key] || key;
    }

    // ==================== PUBLIC API ====================
    return {
        open: function(numSlots, callback) {
            _numSlots = numSlots;
            _callback = callback;
            _selected = [];
            _currentGroupId = null;
            createOverlay();
            loadData();
        },
        close: close,
        selectGroup: selectGroup,
        togglePlayer: togglePlayer,
        toggleAll: toggleAll,
        apply: apply
    };
})();

// Globale Convenience-Funktion

    PlayerPicker.open(numSlots, callback);
}

        function formatValue(val, type) {
            if (type === "num") return val.toString();
            if (type === "str") return val; // Einfacher Text für Pace und Modus
            if (type === "ms3") return (parseInt(val) / 1000).toFixed(3) + "s";
            if (type === "ms") {
                let valInt = parseInt(val);
                let totalSec = Math.floor(valInt / 1000);
                let m = Math.floor(totalSec / 60);
                let s = totalSec % 60;
                let z = Math.floor((valInt % 1000) / 100); 
                if (m > 0) return `${m}:${s.toString().padStart(2,'0')}.${z}`;
                return `${s}.${z}s`;
            }
            if (type === "bool") return parseInt(val) === 1 ? ("👑 " + t('ps_yes', "Ja")) : ("❌ " + t('ps_no', "Nein"));
            return val;

#include "StatsManager.h"
#include "PuckNetwork.h"

Preferences statsPrefs;
uint32_t gameStarts[30] = {0}; // index 1..27 used

// Helfer um MAC Adresse in einen kurzen String (die letzten 6 Zeichen) als Speicher-Key zu wandeln
String macToStr(const uint8_t* mac) {
    char buf[13];
    sprintf(buf, "%02X%02X%02X%02X%02X%02X", mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
    return String(buf);
}

void StatsManager::begin() {
    statsPrefs.begin("pr_stats", false);
    for(int i=1; i<=27; i++) {
        gameStarts[i] = statsPrefs.getUInt(String("g" + String(i)).c_str(), 0);
    }
    statsPrefs.end();
    Serial.println("STATS: System geladen.");
}

void StatsManager::addGameStart(int gameID) {
    if(gameID >= 1 && gameID <= 27) {
        gameStarts[gameID]++;
    }
}

uint32_t StatsManager::getGameStarts(int gameID) {
    if(gameID >= 1 && gameID <= 27) return gameStarts[gameID];
    return 0;
}

uint32_t StatsManager::getPuckClicks(const uint8_t* mac) {
    statsPrefs.begin("pr_stats", true); // read-only
    String key = macToStr(mac).substring(6) + "_c"; 
    uint32_t val = statsPrefs.getUInt(key.c_str(), 0);
    statsPrefs.end();
    return val;
}

uint32_t StatsManager::getPuckTime(const uint8_t* mac) {
    statsPrefs.begin("pr_stats", true);
    String key = macToStr(mac).substring(6) + "_t"; 
    uint32_t val = statsPrefs.getUInt(key.c_str(), 0);
    statsPrefs.end();
    return val;
}

void StatsManager::saveAll() {
    Serial.println("STATS: Speichere Daten in den Flash...");
    statsPrefs.begin("pr_stats", false);
    
    // Spiel-Aufrufe speichern
    for(int i=1; i<=27; i++) {
        statsPrefs.putUInt(String("g" + String(i)).c_str(), gameStarts[i]);
    }
    
    // Aktive Pucks speichern
    PuckInfo* pucks = PuckNetwork::getPucks();
    for(int i=0; i<MAX_PEERS; i++) {
        if(pucks[i].active) {
            String baseKey = macToStr(pucks[i].mac).substring(6);
            statsPrefs.putUInt((baseKey + "_c").c_str(), pucks[i].totalClicks);
            statsPrefs.putUInt((baseKey + "_t").c_str(), pucks[i].totalMinutes);
        }
    }
    
    statsPrefs.end();
    Serial.println("STATS: Speicherung erfolgreich!");
}
