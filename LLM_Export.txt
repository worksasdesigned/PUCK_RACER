    <script>
        if (localStorage.getItem('puck_theme') === 'light') document.body.classList.add('theme-light');

        const numPlayers = parseInt(sessionStorage.getItem('ps_players')   || 1);
        const timeVal    = parseInt(sessionStorage.getItem('ps_time')      || 10);
        const pitlane    = parseInt(sessionStorage.getItem('ps_pitlane')   || 0) === 1;
        const countdown  = parseInt(sessionStorage.getItem('ps_countdown') || 1) === 1;
        const alwyellow   = parseInt(sessionStorage.getItem('ps_alwyellow')  || 0) === 1;
        const statuspuck  = parseInt(sessionStorage.getItem('ps_statuspuck') || 0) === 1;
        const exitTimeVal = parseInt(sessionStorage.getItem('ps_exittime')   || 10);
        const exitbtn     = parseInt(sessionStorage.getItem('ps_exitbtn')    || 0) === 1;

        // Player colors matching C++ PLAYER_COLORS[] — no red/green (reserved for game signals)
        const COLORS = ['#0000FF','#8800FF','#0088FF','#FFFF00','#FF00FF','#00FFFF','#FF8800','#FFFFFF','#FF69B4','#00FFAA'];

        // puck assignment: entryAssign[i] = global puck index for slot i
        let entryAssign    = [];
        let exitAssign     = [];
        let statusPuckAssign = -1;
        let activePuckCount  = 0;

        const delay = ms => new Promise(res => setTimeout(res, ms));

        document.addEventListener('DOMContentLoaded', () => setTimeout(initPage, 400));

        async function initPage() {
            try {
                // Load game & send config
                await fetch('/api/game/load?id=28');
                await fetch(`/api/game/action?cmd=cfg_players&val=${numPlayers}`);
                await fetch(`/api/game/action?cmd=cfg_time&val=${timeVal}`);
                await fetch(`/api/game/action?cmd=cfg_pitlane&val=${pitlane ? 1 : 0}`);
                await fetch(`/api/game/action?cmd=cfg_countdown&val=${countdown ? 1 : 0}`);
                await fetch(`/api/game/action?cmd=cfg_alwyellow&val=${alwyellow ? 1 : 0}`);
                await fetch(`/api/game/action?cmd=cfg_exittime&val=${exitTimeVal}`);
                await fetch(`/api/game/action?cmd=cfg_exitbtn&val=${exitbtn ? 1 : 0}`);
                await fetch('/api/game/action?cmd=setup');

                // Get connected pucks
                let res  = await fetch('/api/status');
                let data = await res.json();
                let active = data.filter(p => p.active);
                activePuckCount = active.length;

                // Build sequential auto-assignment
                for (let i = 0; i < numPlayers; i++) {
                    if (pitlane) {
                        entryAssign[i] = (i * 2     < activePuckCount) ? i * 2     : -1;
                        exitAssign[i]  = (i * 2 + 1 < activePuckCount) ? i * 2 + 1 : -1;
                    } else {
                        entryAssign[i] = (i < activePuckCount) ? i : -1;
                        exitAssign[i]  = -1;
                    }
                }
                // Status puck gets the next puck after all player pucks
                if (statuspuck) {
                    let nextIdx = pitlane ? numPlayers * 2 : numPlayers;
                    statusPuckAssign = (nextIdx < activePuckCount) ? nextIdx : -1;
                }

                // Send assignments to backend
                await sendAssignments();

                // Show pucks with player colors
                await fetch('/api/game/action?cmd=show_track');

                buildArena();
                renderInputs();

                document.getElementById('loading').style.display = 'none';
                document.getElementById('content').style.display = 'block';

                setInterval(pollPuckEffects, 1500);
            } catch (e) {
                console.error(e);
                document.getElementById('loading').innerText = 'Error loading game.';
            }
        }

        async function sendAssignments() {
            for (let i = 0; i < numPlayers; i++) {
                if (entryAssign[i] >= 0)
                    await fetch(`/api/game/action?cmd=cfg_entry&val=${i * 100 + entryAssign[i]}`);
                if (pitlane && exitAssign[i] >= 0)
                    await fetch(`/api/game/action?cmd=cfg_exit&val=${i * 100 + exitAssign[i]}`);
            }
            await fetch(`/api/game/action?cmd=cfg_statuspuck_en&val=${statuspuck ? 1 : 0}`);
            if (statuspuck && statusPuckAssign >= 0)
                await fetch(`/api/game/action?cmd=cfg_statuspuck&val=${statusPuckAssign}`);
        }

        function buildArena() {
            const cols = ['A','B','C','D','E'];
            let tableHtml = `<table class="arena-table">`;
            for (let r = 1; r <= 5; r++) {
                tableHtml += `<tr>`;
                for (let c = 0; c < 5; c++) {
                    tableHtml += `<td id="cell-${cols[c]}${r}"></td>`;
                }
                tableHtml += `</tr>`;
            }
            tableHtml += `</table>`;
            document.getElementById('arenaContainer').innerHTML = tableHtml;

            for (let i = 0; i < numPlayers; i++) {
                let row = (i % 5) + 1;
                if (!pitlane) {
                    // Pucks: A1-A5 (players 0-4), C1-C5 (players 5-9)
                    // Cars:  B1-B5 (players 0-4), D1-D5 (players 5-9)
                    let puckCol = (i < 5) ? 'A' : 'C';
                    let carCol  = (i < 5) ? 'B' : 'D';
                    let puckCell = document.getElementById(`cell-${puckCol}${row}`);
                    let carCell  = document.getElementById(`cell-${carCol}${row}`);
                    if (puckCell) puckCell.innerHTML = `<div class="puck-icon entry-puck" id="puck_vis_entry_${i}">${entryAssign[i] >= 0 ? entryAssign[i]+1 : '?'}</div>`;
                    if (carCell)  carCell.innerHTML  = `<div style="font-size:1.4rem; line-height:1;">🏎️</div>`;
                } else {
                    // Car (mirrored) in A, entry puck in B, exit puck in D
                    if (i >= 5) row = 5; // max 5 rows in pitlane
                    let cellA = document.getElementById(`cell-A${row}`);
                    let cellB = document.getElementById(`cell-B${row}`);
                    let cellD = document.getElementById(`cell-D${row}`);
                    if (cellA) cellA.innerHTML = `<div style="font-size:1.4rem; line-height:1; display:inline-block; transform:scaleX(-1);">🏎️</div>`;
                    if (cellB) cellB.innerHTML = `<div class="puck-icon entry-puck" id="puck_vis_entry_${i}">${entryAssign[i] >= 0 ? entryAssign[i]+1 : '?'}</div>`;
                    if (cellD) cellD.innerHTML = `<div class="puck-icon exit-puck"  id="puck_vis_exit_${i}">${exitAssign[i] >= 0 ? exitAssign[i]+1 : '?'}</div>`;
                }
            }

            // Status puck at E3 (column E is always free)
            if (statuspuck) {
                let cellS = document.getElementById('cell-E3');
                if (cellS) cellS.innerHTML = `<div class="puck-icon entry-puck" id="puck_vis_status" title="Status Puck">${statusPuckAssign >= 0 ? statusPuckAssign + 1 : '?'}</div>`;
            }

            let hintKey  = pitlane ? 'ps_layout_pitlane' : 'ps_layout_simple';
            let hintFall = pitlane ? 'A = Car  |  B = Entry puck  |  D = Exit puck' : 'A = Puck  |  B = Car';
            document.getElementById('layoutHint').innerText = (typeof getTranslation === 'function') ? (getTranslation(hintKey) || hintFall) : hintFall;

            let legKey  = pitlane ? 'ps_legend_pitlane' : 'ps_legend_simple';
            let legFall = pitlane ? 'Entry: Single Chase  |  Exit: Breathe' : 'Each puck glows in player color';
            document.getElementById('arenaLegend').innerText = (typeof getTranslation === 'function') ? (getTranslation(legKey) || legFall) : legFall;
        }

        function renderInputs() {
            let html = '';
            for (let i = 0; i < numPlayers; i++) {
                let col  = COLORS[i % 10];
                let name = sessionStorage.getItem(`ps_name_${i}`) || '';
                let eBadge = entryAssign[i] >= 0 ? `<span class="puck-badge" style="border-color:${col};">🟢 #${entryAssign[i]+1}</span>` : '';
                let xBadge = (pitlane && exitAssign[i] >= 0) ? `<span class="puck-badge" style="opacity:0.6;">⬜ #${exitAssign[i]+1}</span>` : '';
                html += `
                <div class="player-row" style="border-left-color:${col};">
                    <div class="color-box" style="background:${col}; box-shadow:0 0 8px ${col};"></div>
                    <div style="flex:1;">
                        <input type="text" id="name_${i}" value="${name}" placeholder="Player ${i+1}" oninput="saveName(${i})">
                    </div>
                    <div style="display:flex; gap:4px; flex-shrink:0;">${eBadge}${xBadge}</div>
                </div>`;
            }
            document.getElementById('playerList').innerHTML = html;
        }

        function saveName(i) {
            sessionStorage.setItem(`ps_name_${i}`, document.getElementById(`name_${i}`).value);
        }

        function loadFromPicker() {
            openPlayerPicker(numPlayers, function(selected) {
                for (let i = 0; i < numPlayers && i < selected.length; i++) {
                    let inp = document.getElementById(`name_${i}`);
                    if (inp) { inp.value = selected[i].name; sessionStorage.setItem(`ps_name_${i}`, selected[i].name); }
                }
            });
        }

        async function pollPuckEffects() {
            try {
                let r    = await fetch('/api/effects');
                let data = await r.json();
                data.forEach(eff => {
                    let rgb = `rgb(${eff.r},${eff.g},${eff.b})`;
                    for (let i = 0; i < numPlayers; i++) {
                        if (eff.id === entryAssign[i]) {
                            let v = document.getElementById(`puck_vis_entry_${i}`);
                            if (v) { v.style.background = rgb; v.style.border = '2px solid #fff'; v.style.boxShadow = `0 0 15px ${rgb}`; }
                        }
                        if (pitlane && eff.id === exitAssign[i]) {
                            let v = document.getElementById(`puck_vis_exit_${i}`);
                            if (v) { v.style.background = rgb; v.style.border = '2px solid #888'; }
                        }
                    }
                    if (statuspuck && eff.id === statusPuckAssign) {
                        let v = document.getElementById('puck_vis_status');
                        if (v) { v.style.background = rgb; v.style.border = '2px solid #fff'; v.style.boxShadow = `0 0 15px ${rgb}`; }
                    }
                });
            } catch(e) {}
        }

        function toggleLayout() {
            document.getElementById('layoutBody').classList.toggle('hidden');
            document.getElementById('toggleIcon').classList.toggle('rotate');
        }

        function startGame() {
            // Names already saved via oninput; navigate to run screen
            setTimeout(() => { location.href = 'game_pitstop_run.html'; }, 100);
        }
    </script>
</body>
</html>

    <script>
        if (localStorage.getItem('puck_theme') === 'light') document.body.classList.add('theme-light');

        const numPlayers     = parseInt(sessionStorage.getItem('ps_players')  || 1);
        const isPitlane      = parseInt(sessionStorage.getItem('ps_pitlane')  || 0) === 1;
        const pitstopTotalMs = parseInt(sessionStorage.getItem('ps_time')     || 10) * 500;
        const exitTotalMs    = parseInt(sessionStorage.getItem('ps_exittime') || 10) * 500;

        // Colors – no red/green (reserved for game signals)
        const COLORS = ['#0000FF','#8800FF','#0088FF','#FFFF00','#FF00FF','#00FFFF','#FF8800','#FFFFFF','#FF69B4','#00FFAA'];

        let lastData = null;

        function t(key, fallback) {
            if (typeof getTranslation !== 'function') return fallback;
            let v = getTranslation(key);
            return (v && v !== key) ? v : fallback;
        }

        // --- Build player cards ---
        function buildCards() {
            let html = '';
            for (let i = 0; i < numPlayers; i++) {
                let col  = COLORS[i % 10];
                let name = sessionStorage.getItem(`ps_name_${i}`) || `Player ${i+1}`;

                let bars = isPitlane
                    ? `<div class="pc-bars">
                           <div class="bar-row">
                               <div class="bar-lbl" data-i18n="ps_entry">Entry</div>
                               <div class="bar-track"><div class="bar-fill entry" id="bar_entry_${i}" style="width:0%"></div></div>
                               <div class="bar-time" id="time_entry_${i}">--</div>
                           </div>
                           <div class="bar-row">
                               <div class="bar-lbl" data-i18n="ps_exit">Exit</div>
                               <div class="bar-track"><div class="bar-fill exit" id="bar_exit_${i}" style="width:0%"></div></div>
                               <div class="bar-time" id="time_exit_${i}">--</div>
                               <span class="bar-pending" id="pend_${i}" style="display:none;" title="queued">⏳</span>
                           </div>
                       </div>`
                    : `<div class="pc-bars">
                           <div class="bar-row">
                               <div class="bar-lbl" data-i18n="ps_pitstop">Pitstop</div>
                               <div class="bar-track"><div class="bar-fill entry" id="bar_entry_${i}" style="width:0%"></div></div>
                               <div class="bar-time" id="time_entry_${i}">--</div>
                           </div>
                       </div>`;

                html += `
                <div class="player-card active" id="card_${i}" style="border-left-color:${col};" onclick="onCardClick(${i})">
                    <div class="pc-top">
                        <div style="width:12px;height:12px;border-radius:50%;background:${col};flex-shrink:0;box-shadow:0 0 6px ${col};"></div>
                        <div class="pc-name">${name}</div>
                        <div style="text-align:right;">
                            <div class="pc-stops" id="stops_${i}">0</div>
                            <div class="pc-stop-lbl" data-i18n="ps_stops">Stops</div>
                        </div>
                    </div>
                    ${bars}

                    </div>
                </div>`;
            }
            document.getElementById('playerCards').innerHTML = html;
        }
        buildCards();

        // --- Polling ---
        setInterval(update, 400);

        function update() {
            fetch('/api/game/status').then(r => r.json()).then(data => {
                lastData = data;
                render(data);
            }).catch(e => {});
        }

        function fmtMs(ms) {
            if (ms <= 0) return '0.0s';
            return (ms / 1000).toFixed(1) + 's';
        }

        function render(data) {
            document.getElementById('totalStops').innerText = data.total || 0;

            // State label
            const stateMap = { 0:'SETUP', 1:'RUNNING', 2:'PAUSED', 3:'FINISHED' };
            const stateCol = { 0:'var(--warning)', 1:'var(--success)', 2:'var(--warning)', 3:'gold' };
            let stTxt = t(`ps_st_${data.st}`, stateMap[data.st] || 'UNKNOWN');
            document.getElementById('stateInfo').innerText    = stTxt;
            document.getElementById('stateInfo').style.color = stateCol[data.st] || 'var(--warning)';

            // Start button label changes with state
            let btn = document.getElementById('btnStart');

            } else if (data.st === 2) {
                btn.innerText = t('btn_resume', 'RESUME');
            } else {
                btn.innerText = t('btn_pause', 'PAUSE');
            }

            if (!data.slots) return;

                let card     = document.getElementById(`card_${i}`);

                    penBadge.style.display = 'inline';
                    penBadge.innerText = '+' + (slot.pen / 1000).toFixed(1) + 's';

                        entryTxt.innerText  = fmtMs(slot.erem);
                    } else {

                            // Waiting for driver to press exit puck

                            exitEl.style.background = '#FFD700';

                            exitEl.style.background = '';
                            exitEl.style.opacity    = '1';

                }

void Game_Pitstop::setup() {
    Serial.println("GAME: Setup Pitstop");
    initGame();

// ---------------------------------------------------------------------------
void Game_Pitstop::processCommand(String cmd, int value) {

    // --- Configuration (set before 'setup' command) ---
    if      (cmd == "cfg_players")  numPlayers  = constrain(value, 1, MAX_PS_PLAYERS);
    else if (cmd == "cfg_time")     pitstopMs   = (unsigned long)value * 500UL; // value = slider step, 1 step = 0.5s
    else if (cmd == "cfg_pitlane")  pitlaneMode = (value == 1);
    else if (cmd == "cfg_countdown")  showCountdown = (value == 1);
    else if (cmd == "cfg_alwyellow")   alwaysYellow      = (value == 1);
    else if (cmd == "cfg_statuspuck_en") statusPuckEnabled = (value == 1);
    else if (cmd == "cfg_statuspuck")    statusPuckIdx     = value;
    else if (cmd == "cfg_exittime")      pitlaneExitMs     = (unsigned long)value * 500UL;
    else if (cmd == "cfg_exitbtn")       exitButtonMode    = (value == 1);
    else if (cmd == "cfg_entry") {
        int slotIdx = value / 100;
        int puckIdx = value % 100;
        if (slotIdx >= 0 && slotIdx < MAX_PS_PLAYERS) slots[slotIdx].entryPuckIdx = puckIdx;
    }
    else if (cmd == "cfg_exit") {
        int slotIdx = value / 100;
        int puckIdx = value % 100;
        if (slotIdx >= 0 && slotIdx < MAX_PS_PLAYERS) slots[slotIdx].exitPuckIdx = puckIdx;
    }

    // --- Lifecycle ---
    else if (cmd == "setup") initGame();

    else if (cmd == "show_track") {
        for (int i = 0; i < numPlayers; i++) {
            CRGB col = PLAYER_COLORS[i % 10];
            if (slots[i].entryPuckIdx >= 0)
                setPuck(slots[i].entryPuckIdx, EFF_SINGLE_CHASE, col, 20, 220);
            if (pitlaneMode && slots[i].exitPuckIdx >= 0)
                setPuck(slots[i].exitPuckIdx, EFF_BREATHE_MOD4, col, 10, 80);
            delay(10);
        }
        if (statusPuckEnabled && statusPuckIdx >= 0)
            setPuck(statusPuckIdx, EFF_RAINBOW, CRGB::Black, 10, 220);
    }

    else if (cmd == "start") {
        if (gameState == PS_SETUP || gameState == PS_FINISHED) {
            gameState = PS_RUNNING;
            totalStops = 0;
            lastStatusState = 0;
            for (int i = 0; i < numPlayers; i++) {
                slots[i].stopCount    = 0;
                slots[i].entryRunning = false;
                slots[i].exitRunning  = false;
                slots[i].exitPending  = false;
                slots[i].entryDoneAt            = 0;
                slots[i].exitDoneAt             = 0;
                slots[i].penaltyMs              = 0;
                slots[i].exitWaitingForButton   = false;
                slots[i].exitPoliceAt           = 0;
                if (slots[i].isActive) initSlotPucks(i);
            }
            updateStatusPuck();
        }
    }

    else if (cmd == "pause") {
        if (gameState == PS_RUNNING) {
            gameState    = PS_PAUSED;
            pauseStartMs = millis();
            // Show red breathe on all active pucks
            for (int i = 0; i < numPlayers; i++) {
                if (!slots[i].isActive) continue;
                if (slots[i].entryPuckIdx >= 0)
                    setPuck(slots[i].entryPuckIdx, EFF_BREATHE_MOD4, CRGB::Red, 10, 200);
                if (pitlaneMode && slots[i].exitPuckIdx >= 0)
                    setPuck(slots[i].exitPuckIdx,  EFF_BREATHE_MOD4, CRGB::Red, 10, 200);
            }
            lastStatusState = 0;
            updateStatusPuck();
        } else if (gameState == PS_PAUSED) {
            // Compensate all running timers for the paused duration
            unsigned long pausedMs = millis() - pauseStartMs;
            for (int i = 0; i < numPlayers; i++) {
                PitstopSlot& s = slots[i];
                if (s.entryRunning)        s.entryStart   += pausedMs;
                if (s.exitRunning)         s.exitStart    += pausedMs;
                if (s.entryDoneAt  > 0)   s.entryDoneAt  += pausedMs;
                if (s.exitDoneAt   > 0)   s.exitDoneAt   += pausedMs;
                if (s.exitPoliceAt > 0)   s.exitPoliceAt += pausedMs;
            }
            gameState = PS_RUNNING;
            // Restore correct visuals for each slot
            for (int i = 0; i < numPlayers; i++) {
                if (!slots[i].isActive) continue;
                restoreSlotVisuals(i);
            }
            lastStatusState = 0;
            updateStatusPuck();
        }
    }

    else if (cmd == "stop") {
        gameState = PS_FINISHED;
        for (int i = 0; i < numPlayers; i++) {
            slots[i].entryRunning         = false;
            slots[i].exitRunning          = false;
            slots[i].exitWaitingForButton = false;
            slots[i].exitPoliceAt         = 0;
            if (slots[i].entryPuckIdx >= 0)
                setPuck(slots[i].entryPuckIdx, EFF_STATUS, CRGB::Green, 0, 85);
            if (pitlaneMode && slots[i].exitPuckIdx >= 0)
                setPuck(slots[i].exitPuckIdx, EFF_STATUS, CRGB::Green, 0, 85);
        }
        if (statusPuckEnabled && statusPuckIdx >= 0)
            setPuck(statusPuckIdx, EFF_RAINBOW, CRGB::Black, 10, 220);
        lastStatusState = 0;
    }

    else if (cmd == "reset") {
        // Full reset including puck assignments → go back to auto-assign on next setup
        for (int i = 0; i < MAX_PS_PLAYERS; i++) {
            slots[i].entryPuckIdx = -1;
            slots[i].exitPuckIdx  = -1;
        }
        initGame();
    }

    else if (cmd == "exit") {
        CommandPacket cp; memset(&cp, 0, sizeof(cp));
        cp.cmd = CMD_EFFECT; cp.effectID = EFF_STATUS;
        cp.r = 0; cp.g = 255; cp.b = 0; cp.extra = 255;
        PuckNetwork::broadcast(cp);
        gameState = PS_SETUP;
        lastStatusState = 0;
    }

    // --- In-game commands ---
    else if (cmd == "trigger") {
        // Manually trigger a player's pitstop entry (trainer button on run screen)
        int slotIdx = value;
        if (slotIdx >= 0 && slotIdx < numPlayers && gameState == PS_RUNNING) {
            if (!slots[slotIdx].entryRunning && slots[slotIdx].isActive)
                triggerEntry(slotIdx);
        }
    }

    else if (cmd == "penalty") {
        // Add N seconds of penalty to a slot: value = slotIdx * 100 + seconds
        int slotIdx = value / 100;
        int seconds = value % 100;
        if (slotIdx >= 0 && slotIdx < numPlayers) {
            unsigned long addMs = (unsigned long)(seconds * 1000);
            slots[slotIdx].penaltyMs += (int)addMs;
            // If countdown is currently running, extend duration and resync puck effect
            if (slots[slotIdx].entryRunning) {
                slots[slotIdx].entryDuration += addMs;
                if (showCountdown && slots[slotIdx].entryPuckIdx >= 0) {
                    unsigned long elapsed = millis() - slots[slotIdx].entryStart;
                    unsigned long rem = (slots[slotIdx].entryDuration > elapsed)
                                        ? slots[slotIdx].entryDuration - elapsed : 0;
                    int spd = constrain((int)(rem / 35), 1, 1500);
                    setPuck(slots[slotIdx].entryPuckIdx, EFF_COUNTDOWN, PLAYER_COLORS[slotIdx % 10], spd, 255);
                }
            }
        }
    }

    else if (cmd == "clr_pen") {
        int slotIdx = value;
        if (slotIdx >= 0 && slotIdx < numPlayers)
            slots[slotIdx].penaltyMs = 0;
    }

    else if (cmd == "toggle_active") {
        int slotIdx = value;
        if (slotIdx >= 0 && slotIdx < numPlayers) {
            slots[slotIdx].isActive = !slots[slotIdx].isActive;
            if (gameState == PS_RUNNING) {
                if (slots[slotIdx].isActive) {
                    initSlotPucks(slotIdx);
                } else {
                    slots[slotIdx].entryRunning         = false;
                    slots[slotIdx].exitRunning          = false;
                    slots[slotIdx].exitWaitingForButton = false;
                    slots[slotIdx].exitPoliceAt         = 0;
                    CRGB playerCol = PLAYER_COLORS[slotIdx % 10];
                    if (slots[slotIdx].entryPuckIdx >= 0)
                        setPuck(slots[slotIdx].entryPuckIdx, EFF_STATUS, playerCol, 0, 85);
                    if (pitlaneMode && slots[slotIdx].exitPuckIdx >= 0)
                        setPuck(slots[slotIdx].exitPuckIdx,  EFF_STATUS, playerCol, 0, 85);
                }
                lastStatusState = 0;
                updateStatusPuck();
            }
        }
    }
}

// ---------------------------------------------------------------------------
void Game_Pitstop::initGame() {
    PuckInfo* netPucks = PuckNetwork::getPucks();
    int activePucks[MAX_PEERS];
    int activePuckCount = 0;
    for (int i = 0; i < MAX_PEERS; i++) {
        if (netPucks[i].active) activePucks[activePuckCount++] = i;
    }

    totalStops = 0;

    for (int i = 0; i < MAX_PS_PLAYERS; i++) {
        // Reset runtime state
        slots[i].entryRunning = false;
        slots[i].exitRunning  = false;
        slots[i].exitPending  = false;
        slots[i].stopCount    = 0;
        slots[i].isActive     = true;
        slots[i].penaltyMs    = 0;
        slots[i].entryStart           = 0;
        slots[i].entryDuration        = 0;
        slots[i].entryDoneAt          = 0;
        slots[i].exitStart            = 0;
        slots[i].exitDoneAt           = 0;
        slots[i].exitWaitingForButton = false;
        slots[i].exitPoliceAt         = 0;

        // Auto-assign pucks only if not yet configured via cfg_entry / cfg_exit
        if (slots[i].entryPuckIdx < 0 && i < numPlayers) {
            if (pitlaneMode) {
                slots[i].entryPuckIdx = (i * 2     < activePuckCount) ? activePucks[i * 2]     : -1;
                slots[i].exitPuckIdx  = (i * 2 + 1 < activePuckCount) ? activePucks[i * 2 + 1] : -1;
            } else {
                slots[i].entryPuckIdx = (i < activePuckCount) ? activePucks[i] : -1;
                slots[i].exitPuckIdx  = -1;
            }
        }
    }

    gameState = PS_SETUP;
}

// ---------------------------------------------------------------------------
void Game_Pitstop::triggerEntry(int slotIdx) {
    PitstopSlot& slot = slots[slotIdx];
    unsigned long dur = pitstopMs + (unsigned long)slot.penaltyMs;
    slot.entryRunning  = true;
    slot.entryStart    = millis();
    slot.entryDuration = dur;
    slot.entryDoneAt   = 0;

    if (slot.entryPuckIdx >= 0) {
        CRGB col = PLAYER_COLORS[slotIdx % 10];
        if (showCountdown) {
            int spd = constrain((int)(dur / 35), 1, 1500);
            setPuck(slot.entryPuckIdx, EFF_COUNTDOWN, col, spd, 255);
        } else {
            setPuck(slot.entryPuckIdx, EFF_STATIC, col, 0, 255);
        }
        sendSound(slot.entryPuckIdx, 50);
    }
}

// ---------------------------------------------------------------------------
void Game_Pitstop::completeEntry(int slotIdx) {
    PitstopSlot& slot = slots[slotIdx];
    slot.entryRunning = false;

    if (!pitlaneMode) {
        // Simple mode: entry done = pitstop complete
        totalStops++;
        slot.stopCount++;
        slot.penaltyMs  = 0;
        slot.entryDoneAt = millis();   // trigger 2s idle-return in loop()
        if (slot.entryPuckIdx >= 0) {
            sendSequence(slot.entryPuckIdx, SEQ_DINGDONG);
            setPuck(slot.entryPuckIdx, EFF_WIN, CRGB::Green, 0, 255);
        }
    } else {
        // Pitlane mode: entry done → start or queue exit countdown
        // Entry puck goes back to idle (waiting for next car)
        if (slot.entryPuckIdx >= 0) {
            CRGB col = alwaysYellow ? CRGB::Yellow : PLAYER_COLORS[slotIdx % 10];
            setPuck(slot.entryPuckIdx, EFF_BREATHE_MOD4, col, 10, 200);
        }

        if (!slot.exitRunning && !slot.exitWaitingForButton && !slot.exitPoliceAt) {
            startExitCountdown(slotIdx);
        } else {
            // Exit puck still busy → queue this car
            slot.exitPending = true;
        }
    }
}

// ---------------------------------------------------------------------------
void Game_Pitstop::completeExit(int slotIdx) {
    PitstopSlot& slot = slots[slotIdx];
    slot.exitRunning          = false;
    slot.exitWaitingForButton = false;
    slot.exitPoliceAt         = 0;
    totalStops++;
    slot.stopCount++;
    slot.penaltyMs = 0;

    if (slot.exitPuckIdx >= 0) {
        sendSequence(slot.exitPuckIdx, SEQ_DINGDONG);
        setPuck(slot.exitPuckIdx, EFF_WIN, CRGB::Green, 0, 255);
    }

    if (slot.exitPending) {
        slot.exitPending = false;
        startExitCountdown(slotIdx);
    } else {
        slot.exitDoneAt = millis();  // trigger 2s idle-return in loop()
    }
}

// ---------------------------------------------------------------------------
void Game_Pitstop::initSlotPucks(int slotIdx) {
    CRGB col = alwaysYellow ? CRGB::Yellow : PLAYER_COLORS[slotIdx % 10];
    if (slots[slotIdx].entryPuckIdx >= 0)
        setPuck(slots[slotIdx].entryPuckIdx, EFF_BREATHE_MOD4, col, 10, 200);
    if (pitlaneMode && slots[slotIdx].exitPuckIdx >= 0)
        setPuck(slots[slotIdx].exitPuckIdx,  EFF_BREATHE_MOD4, col, 10, 80);
}

// ---------------------------------------------------------------------------
void Game_Pitstop::loop() {
    updateStatusPuck();
    if (gameState != PS_RUNNING) return;

    unsigned long now = millis();

    for (int i = 0; i < numPlayers; i++) {
        PitstopSlot& slot = slots[i];
        if (!slot.isActive) continue;

        // Entry countdown completed?
        if (slot.entryRunning && (now - slot.entryStart >= slot.entryDuration))
            completeEntry(i);

        // Exit countdown completed? (pitlane mode)
        if (pitlaneMode && slot.exitRunning && (now - slot.exitStart >= pitlaneExitMs)) {
            if (exitButtonMode) {
                // Wait for driver to press exit puck
                slot.exitRunning          = false;
                slot.exitWaitingForButton = true;
                if (slot.exitPuckIdx >= 0)
                    setPuck(slot.exitPuckIdx, EFF_SINGLE_CHASE, CRGB::Yellow, 20, 220);
            } else {
                completeExit(i);
            }
        }

        // Police timer: 3s after early press → complete exit
        if (pitlaneMode && slot.exitPoliceAt > 0 && (now - slot.exitPoliceAt >= 3000))
            completeExit(i);

        // Return entry puck to idle 2s after simple-mode pitstop
        if (!pitlaneMode && !slot.entryRunning && slot.entryDoneAt > 0
            && (now - slot.entryDoneAt >= 2000)) {
            slot.entryDoneAt = 0;
            if (slot.entryPuckIdx >= 0) {
                CRGB col = alwaysYellow ? CRGB::Yellow : PLAYER_COLORS[i % 10];
                setPuck(slot.entryPuckIdx, EFF_BREATHE_MOD4, col, 10, 200);
            }
        }

        // Return exit puck to idle 2s after exit completion (pitlane)
        if (pitlaneMode && !slot.exitRunning && !slot.exitPending
            && slot.exitDoneAt > 0 && (now - slot.exitDoneAt >= 2000)) {
            slot.exitDoneAt = 0;
            if (slot.exitPuckIdx >= 0) {
                CRGB col = alwaysYellow ? CRGB::Yellow : PLAYER_COLORS[i % 10];
                setPuck(slot.exitPuckIdx, EFF_BREATHE_MOD4, col, 10, 80);
            }
        }
    }
}

// ---------------------------------------------------------------------------
void Game_Pitstop::handleEvent(int puckIndex, EventPacket event) {
    if (gameState != PS_RUNNING) return;
    if (event.type != EVT_BTN_CLICK) return;

    for (int i = 0; i < numPlayers; i++) {
        if (!slots[i].isActive) continue;

        // Entry puck: trigger pitstop
        if (puckIndex == slots[i].entryPuckIdx && !slots[i].entryRunning) {
            triggerEntry(i);
            return;
        }

        // Exit puck button (exitButtonMode only)
        if (exitButtonMode && pitlaneMode && puckIndex == slots[i].exitPuckIdx) {
            if (slots[i].exitWaitingForButton) {
                // On time (yellow) → complete normally
                slots[i].exitWaitingForButton = false;
                completeExit(i);
                return;
            } else if (slots[i].exitRunning) {
                // Too early (still red) → police penalty
                slots[i].exitRunning  = false;
                slots[i].exitPoliceAt = millis();
                if (slots[i].exitPuckIdx >= 0)
                    setPuck(slots[i].exitPuckIdx, EFF_POLICE, CRGB::Black, 0, 255);
                return;
            }
        }
    }
}

// ---------------------------------------------------------------------------
String Game_Pitstop::getStatusJSON() {
    unsigned long now = millis();
    String json = "{";
    json += "\"st\":"    + String(gameState)            + ",";
    json += "\"total\":" + String(totalStops)           + ",";
    json += "\"pit\":"   + String(pitlaneMode ? 1 : 0)  + ",";
    json += "\"slots\":[";
    for (int i = 0; i < numPlayers; i++) {
        if (i > 0) json += ",";
        PitstopSlot& s = slots[i];

        long erem = 0, xrem = 0;
        if (s.entryRunning) {
            erem = (long)s.entryDuration - (long)(now - s.entryStart);
            if (erem < 0) erem = 0;
        }
        if (pitlaneMode && s.exitRunning) {
            xrem = (long)pitlaneExitMs - (long)(now - s.exitStart);
            if (xrem < 0) xrem = 0;
        }

        json += "{";
        json += "\"entr\":"  + String(s.entryRunning ? 1 : 0)               + ",";
        json += "\"exit\":"  + String((pitlaneMode && s.exitRunning) ? 1 : 0) + ",";
        json += "\"pend\":"  + String(s.exitPending ? 1 : 0)                + ",";
        json += "\"erem\":"  + String(erem)                                  + ",";
        json += "\"xrem\":"  + String(xrem)                                  + ",";
        json += "\"sc\":"    + String(s.stopCount)                           + ",";
        json += "\"act\":"   + String(s.isActive ? 1 : 0)                   + ",";
        json += "\"ewfb\":"  + String(s.exitWaitingForButton ? 1 : 0)       + ",";
        json += "\"pen\":"   + String(s.penaltyMs);
        json += "}";
    }
    json += "]}";
    return json;
}

// ---------------------------------------------------------------------------
void Game_Pitstop::startExitCountdown(int slotIdx) {
    PitstopSlot& slot = slots[slotIdx];
    slot.exitRunning          = true;
    slot.exitStart            = millis();
    slot.exitDoneAt           = 0;
    slot.exitWaitingForButton = false;
    slot.exitPoliceAt         = 0;
    if (slot.exitPuckIdx >= 0) {
        if (exitButtonMode) {
            int spd = constrain((int)(pitlaneExitMs / 35), 1, 1500);
            setPuck(slot.exitPuckIdx, EFF_COUNTDOWN, CRGB::Red, spd, 220);
        } else if (showCountdown) {
            CRGB col = PLAYER_COLORS[slotIdx % 10];
            int spd = constrain((int)(pitlaneExitMs / 35), 1, 1500);
            setPuck(slot.exitPuckIdx, EFF_COUNTDOWN, col, spd, 200);
        } else {
            CRGB col = PLAYER_COLORS[slotIdx % 10];
            setPuck(slot.exitPuckIdx, EFF_DOUBLE_CHASE, col, 20, 200);
        }
        sendSound(slot.exitPuckIdx, 50);
    }
}

// ---------------------------------------------------------------------------
void Game_Pitstop::updateStatusPuck() {
    if (!statusPuckEnabled || statusPuckIdx < 0) return;
    if (gameState != PS_RUNNING && gameState != PS_PAUSED) return;

    int newState; // 1=green, 2=red
    if (gameState == PS_PAUSED) {
        newState = 2;
    } else {
        // Green if at least one active slot has its entry free (pitlane open)
        bool anyOpen = false;
        for (int i = 0; i < numPlayers; i++) {
            if (slots[i].isActive && !slots[i].entryRunning) { anyOpen = true; break; }
        }
        newState = anyOpen ? 1 : 2;
    }

    if (newState == lastStatusState) return; // avoid re-sending identical command
    lastStatusState = newState;
    CRGB col = (newState == 1) ? CRGB::Green : CRGB::Red;
    setPuck(statusPuckIdx, EFF_DOUBLE_CHASE, col, 20, 220);
}

// ---------------------------------------------------------------------------
// Re-apply the correct visual for a slot after pause/resume
void Game_Pitstop::restoreSlotVisuals(int slotIdx) {
    PitstopSlot& slot = slots[slotIdx];
    CRGB idleCol = alwaysYellow ? CRGB::Yellow : PLAYER_COLORS[slotIdx % 10];
    CRGB plyCol  = PLAYER_COLORS[slotIdx % 10];

    // --- Entry puck ---
    if (slot.entryPuckIdx >= 0) {
        if (slot.entryRunning) {
            if (showCountdown) {
                unsigned long rem = 0;
                unsigned long elapsed = millis() - slot.entryStart;
                if (elapsed < slot.entryDuration) rem = slot.entryDuration - elapsed;
                int spd = constrain((int)(rem / 35), 1, 1500);
                setPuck(slot.entryPuckIdx, EFF_COUNTDOWN, plyCol, spd, 255);
            } else {
                setPuck(slot.entryPuckIdx, EFF_STATIC, plyCol, 0, 255);
            }
        } else {
            setPuck(slot.entryPuckIdx, EFF_BREATHE_MOD4, idleCol, 10, 200);
        }
    }

    // --- Exit puck (pitlane mode) ---
    if (pitlaneMode && slot.exitPuckIdx >= 0) {
        if (slot.exitWaitingForButton) {
            setPuck(slot.exitPuckIdx, EFF_SINGLE_CHASE, CRGB::Yellow, 20, 220);
        } else if (slot.exitRunning) {
            unsigned long elapsed = millis() - slot.exitStart;
            if (exitButtonMode) {
                unsigned long rem = (elapsed < pitlaneExitMs) ? pitlaneExitMs - elapsed : 0;
                int spd = constrain((int)(rem / 35), 1, 1500);
                setPuck(slot.exitPuckIdx, EFF_COUNTDOWN, CRGB::Red, spd, 220);
            } else if (showCountdown) {
                unsigned long rem = (elapsed < pitlaneExitMs) ? pitlaneExitMs - elapsed : 0;
                int spd = constrain((int)(rem / 35), 1, 1500);
                setPuck(slot.exitPuckIdx, EFF_COUNTDOWN, plyCol, spd, 200);
            } else {
                setPuck(slot.exitPuckIdx, EFF_DOUBLE_CHASE, plyCol, 20, 200);
            }
        } else {
            setPuck(slot.exitPuckIdx, EFF_BREATHE_MOD4, idleCol, 10, 80);
        }
    }
}

}

}

    PuckNetwork::sendToPuck(PuckNetwork::getPucks()[globalIdx].mac, seq);
}

    int exitPuckIdx  = -1;      // -1 in simple mode (no pitlane)

    // Entry countdown
    bool entryRunning = false;
    unsigned long entryStart    = 0;
    unsigned long entryDuration = 0;
    unsigned long entryDoneAt   = 0;  // timestamp when entry completed → return to idle after 2s

    // Exit countdown (pitlane mode only)
    bool exitRunning          = false;
    unsigned long exitStart   = 0;
    bool exitPending          = false;  // next exit queued (new car in, old car still at exit)
    unsigned long exitDoneAt  = 0;     // timestamp when exit completed → return to idle after 2s
    bool exitWaitingForButton = false;  // exit timer done, waiting for driver to press puck
    unsigned long exitPoliceAt = 0;    // when police effect started (0 = not active)

    int  stopCount = 0;
    bool isActive  = true;
    int  penaltyMs = 0;             // accumulated penalty time (ms)

    void   loop()     override;
    void   handleEvent(int puckIndex, EventPacket event) override;
    String getName()  override { return "Pitstop"; }

    String getStatusJSON() override;
    void   processCommand(String cmd, int value) override;

private:
    static const int MAX_PS_PLAYERS = 10;

    // --- Config ---
    int           numPlayers       = 1;
    unsigned long pitstopMs        = 5000;  // target pitstop duration in ms
    bool          pitlaneMode      = false; // 2 pucks per player
    bool          showCountdown    = true;  // show EFF_COUNTDOWN on pucks
    bool          alwaysYellow     = false; // idle pucks show yellow instead of player color
    bool          statusPuckEnabled = false; // 1 extra puck shows pitlane open/closed state
    int           statusPuckIdx    = -1;    // global puck index of the status puck
    unsigned long pitlaneExitMs   = 5000;  // exit countdown duration (separate from pitstopMs)
    bool          exitButtonMode  = false;  // driver must press exit puck to confirm pitlane done

    // --- State ---
    PS_GameState  gameState    = PS_SETUP;
    unsigned long pauseStartMs = 0;         // millis() when game was paused
    PitstopSlot   slots[MAX_PS_PLAYERS];
    int           totalStops   = 0;
    int           lastStatusState = 0;      // 0=none, 1=green, 2=red (avoid re-sending same command)

    // Player colors — no red/green (reserved for game signals: stop/go, win/fail)

    unsigned long now = millis();
    IPAddress clientIP = req->client()->remoteIP();

        activeClientIP = clientIP;
        lastActivityTime = now;
        activeClientName = clientIP.toString();
        return true;
    }
    if (clientIP == activeClientIP) {
        lastActivityTime = now;
        return true;
    }
    return false;
}

    // EXTENDED PLAYER STATISTICS (CSV LOGGING)
    // =========================================================================
    // Each line in the /s_<ID>.csv has exactly this format (8 columns):
    // Timestamp;GameID;Score;Ex1;Ex2;Ex3;Ex4;Ex5
    // 
    // DATA STRUCTURE DEPENDING ON GAME-ID:
    // GameID  1 (Shuttle Run)   : Score = Time (ms)             | Ex1 = Rounds
    // GameID  3 (Countdown)     : Score = Time (ms)             | Ex1 = Clicks
    // GameID  4 (Simon Says)    : Score = Time (ms)             | Ex1 = Level
    // GameID  8 (Bomb Defusal)  : Score = Avg Deviation (ms)    | Ex1 = Best Round (ms)
    // GameID 13 (React 2P)      : Score = Reaction Time (ms)    | Ex1 = 1 (Win) or 0 (Loss)
    // GameID 14 (T-Test)        : Score = Time (ms)             | Ex1 = 0
    // GameID 15 (Target)        : Score = Hits                  | Ex1 = Time (ms)
    // GameID 21 (Batak Pro)     : Score = Avg Reaction (ms)     | Ex1 = Hits | Ex2 = Misses | Ex3 = Time (s) | Ex4 = Level | Ex5 = Pucks
    // GameID 24 (Whac-A-Mole)   : Score = Hits                  | Ex1 = Fails
    // GameID 25 (Katzenreflexe) : Score = Avg Reaction (ms)     | Ex1 = Points
    // =========================================================================
    // API: Checks if a CSV already exists for a list of player IDs (e.g., "1,2,3")

    server.on("/api/stats/check", HTTP_GET, [](AsyncWebServerRequest *req){
        if(!req->hasParam("pids")) {
            req->send(400, "application/json", "{\"error\":\"missing pids\"}");
            return;
        }
        
        String pidsParam = req->getParam("pids")->value();
        String json = "{";
        int start = 0;
        bool first = true;
        
        // Komma-separierte Liste aufsplitten und abarbeiten
        while(start < pidsParam.length()) {
            int idx = pidsParam.indexOf(',', start);
            if(idx == -1) idx = pidsParam.length();
            
            String pid = pidsParam.substring(start, idx);
            start = idx + 1; // Für den nächsten Durchlauf
            pid.trim();
            
            if(pid.length() > 0) {
                // Prüfen ob die Datei existiert
                String path = "/s_" + pid + ".csv";
                bool exists = LittleFS.exists(path);

                if(!first) json += ",";
                json += "\"" + pid + "\":" + (exists ? "true" : "false");
                first = false;
            }
        }
        json += "}";
        req->send(200, "application/json", json);
    });

            if (req->hasParam("pid", true) && req->hasParam("game", true) && req->hasParam("score", true) && req->hasParam("ts", true)) {
                String pid = req->getParam("pid", true)->value();
                if (pid == "0" || pid == "") {

                }
                
                String ts = req->getParam("ts", true)->value();
                String game = req->getParam("game", true)->value();
                String score = req->getParam("score", true)->value();
                
                // Abwärtskompatibilität: Nimmt "ex1", ansonsten "extra" (für die anderen Spiele)
                String ex1 = "0";
                if (req->hasParam("ex1", true)) ex1 = req->getParam("ex1", true)->value();
                else if (req->hasParam("extra", true)) ex1 = req->getParam("extra", true)->value();

                String ex2 = req->hasParam("ex2", true) ? req->getParam("ex2", true)->value() : "0";
                String ex3 = req->hasParam("ex3", true) ? req->getParam("ex3", true)->value() : "0";

                String ex5 = req->hasParam("ex5", true) ? req->getParam("ex5", true)->value() : "0";

                // Format: Timestamp;GameID;Score;Ex1;Ex2;Ex3;Ex4;Ex5
                String line = ts + ";" + game + ";" + score + ";" + ex1 + ";" + ex2 + ";" + ex3 + ";" + ex4 + ";" + ex5 + "\n";
                String filename = "/s_" + pid + ".csv";
                
                File f = LittleFS.open(filename, "a"); // Append (Anhängen)
                if (f) {
                    f.print(line);
                    f.close();
                    req->send(200, "text/plain", "OK");
